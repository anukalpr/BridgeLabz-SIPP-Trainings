DEFINE Interface Crop

METHOD waterRequirement()
METHOD harvesting()
}

{

{

}

{

}
}

{
METHOD waterRequirement()
{
PRINT "Wheat requires about 450 mm water per season."
}
METHOD harvesting()
{
PRINT "Wheat is harvested using sickles or harvesters."
}
}
